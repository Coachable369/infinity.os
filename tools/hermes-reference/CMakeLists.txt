cmake_minimum_required(VERSION 3.20)
project(InfinityHermesReference LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(UPSTREAM "${REPO}/build/voice-whisper-src")
if(NOT EXISTS "${UPSTREAM}/examples/talk-llama/llama.h")
  message(FATAL_ERROR "Existing pinned Whisper source with upstream llama reference is required")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GGML_CCACHE OFF CACHE BOOL "" FORCE)
add_subdirectory("${UPSTREAM}/ggml" ggml)
file(GLOB LLAMA_SOURCES "${UPSTREAM}/examples/talk-llama/llama*.cpp")
add_library(hermes_reference_llama STATIC ${LLAMA_SOURCES}
  "${UPSTREAM}/examples/talk-llama/unicode.cpp"
  "${UPSTREAM}/examples/talk-llama/unicode-data.cpp")
target_include_directories(hermes_reference_llama PUBLIC "${UPSTREAM}/examples/talk-llama"
  PRIVATE "${UPSTREAM}/ggml/src")
target_link_libraries(hermes_reference_llama PUBLIC ggml)
add_executable(hermes-reference main.cpp)
target_link_libraries(hermes-reference PRIVATE hermes_reference_llama)
